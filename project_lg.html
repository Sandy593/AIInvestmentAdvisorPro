Project Overview

LangGraph-Powered Multi-Agent System for Investment Analysis

🎯 Problem Statement

Traditional investment research requires extensive manual analysis across multiple data sources, often leading to missed opportunities and biased decision-making. Retail investors lack access to sophisticated research tools available to institutions.


Key Challenges:

  • Manual analysis takes 2-4 hours per investment decision
  • Limited coverage - typically analyze 10-20 assets manually
  • Bias towards well-known stocks (FAANG syndrome)
  • Difficulty discovering emerging opportunities
  • Lack of real-time validation

💡 Solution: LangGraph Multi-Agent System

Built a 14-agent autonomous system using LangGraph that performs end-to-end investment research:

  • LangGraph StateGraph orchestration for agent coordination
  • Conditional routing based on asset types (Stocks/Cryptos/ETFs)
  • Parallel agent execution for complex analysis tasks
  • State persistence with MemorySaver checkpointing
  • Tool augmentation with OpenAI GPT-4 and Yahoo Finance APIs
  • Dynamic discovery without hardcoded asset lists
  • Real-time market data validation
  • Comprehensive report generation with actionable insights

from langgraph.graph import StateGraph, END
from typing import TypedDict, List, Dict, Annotated
from operator import add

class AgentState(TypedDict):
  """Global state shared across all 14 agents"""
  # Input parameters
  user_request: str
  asset_type: str  # "Trending Stocks", "Trending Cryptos", "Trending ETFs"
  sectors: List[str]

  # Discovery phase outputs
  discovered_assets: List[Dict]
  sector_discoveries: Dict[str, List]

  # Analysis phase outputs
  company_analyses: List[Dict]
  sentiment_analyses: List[Dict]

  # Validation phase outputs
  validated_assets: List[Dict]
  market_data: Dict[str, Dict]

  # Emerging opportunities outputs
  emerging_stocks: List[Dict]
  emerging_cryptos: List[Dict]
  emerging_etfs: List[Dict]

  # Memory and control
  exclude_symbols: List[str]
  session_id: str
  messages: Annotated[List[str], add]  # Agent action log

def create_investment_advisor_graph():
  """Create the complete 14-agent LangGraph workflow"""
  
  # Initialize StateGraph
  workflow = StateGraph(AgentState)
  
  # Add all 14 agent nodes
  workflow.add_node("discovery_router", trending_assets_discovery_agent)
  workflow.add_node("sector_specialist", sector_stock_specialist_agent)
  workflow.add_node("crypto_discovery", crypto_discovery_agent)
  workflow.add_node("etf_strategy", etf_strategy_agent)
  workflow.add_node("company_intelligence", company_intelligence_agent)
  workflow.add_node("sentiment_analysis", market_sentiment_agent)
  workflow.add_node("market_validation", market_data_validation_agent)
  workflow.add_node("opportunity_hunter", emerging_opportunities_hunter)
  workflow.add_node("penny_stocks", penny_stock_hunter_agent)
  workflow.add_node("moonshot_cryptos", moonshot_crypto_agent)
  workflow.add_node("innovative_etfs", innovative_etf_scout_agent)
  workflow.add_node("report_generation", report_generation_agent)
  workflow.add_node("database_persist", database_persistence_agent)
  workflow.add_node("communication", communication_agent)
  
  # Set entry point
  workflow.set_entry_point("discovery_router")
  
  # Add conditional routing for asset types
  workflow.add_conditional_edges(
    "discovery_router",
    route_asset_type,  # Routing function
    {
      "sector_specialist": "sector_specialist",
      "crypto_discovery": "crypto_discovery",
      "etf_strategy": "etf_strategy"
    }
  )
  
  # Add conditional routing for emerging opportunities
  workflow.add_conditional_edges(
    "opportunity_hunter",
    route_emerging_type,
    {
      "penny_stocks": "penny_stocks",
      "moonshot_cryptos": "moonshot_cryptos",
      "innovative_etfs": "innovative_etfs"
    }
  )
  
  # Compile with checkpointing
  memory = MemorySaver()
  app = workflow.compile(checkpointer=memory)
  
  return app

🔄 LangGraph Routing Functions

def route_asset_type(state: AgentState) -> str:
  """Route based on asset type selection"""
  asset_type = state.get("asset_type", "")
  
  if asset_type == "Trending Stocks":
    return "sector_specialist"
  elif asset_type == "Trending Cryptos":
    return "crypto_discovery"
  elif asset_type == "Trending ETFs":
    return "etf_strategy"
  else:
    return "sector_specialist"  # Default

def route_emerging_type(state: AgentState) -> str:
  """Route to appropriate emerging opportunity agent"""
  # Dynamic routing based on current state
  if state.get("asset_type") == "Trending Stocks":
    return "penny_stocks"
  elif state.get("asset_type") == "Trending Cryptos":
    return "moonshot_cryptos"
  else:
    return "innovative_etfs"

def should_analyze_companies(state: AgentState) -> str:
  """Conditional analysis for stocks only"""
  if state.get("asset_type") == "Trending Stocks":
    return "analyze"
  return "skip"

# LangGraph Agent Node Example
def sector_stock_specialist_agent(state: AgentState) -> AgentState:
  """Discovers 8-12 stocks per sector using LangGraph state"""
  state["messages"].append("📈 Sector Specialist: Starting discovery")
  
  sector_discoveries = {}
  for sector in state["sectors"]:
    # Step 1: Dynamic context selection
    market_contexts = [
      "focus on emerging companies with recent catalysts",
      "include a mix of established and growth companies",
      "emphasize companies with strong recent earnings"
    ]
    context = random.choice(market_contexts)
    
    # Step 2: Temperature variation for diversity
    temperature = random.uniform(0.3, 0.7)
    
    # Step 3: Tool usage - OpenAI API
    discoveries = call_openai_for_sector(sector, context, temperature)
    
    # Step 4: Update state
    sector_discoveries[sector] = discoveries
    state["messages"].append(f"  Found {len(discoveries)} stocks in {sector}")
  
  # Step 5: Update global state
  state["sector_discoveries"] = sector_discoveries
  state["discovered_assets"] = flatten(sector_discoveries.values())
  
  return state

def generate_discovery_prompt(state: AgentState, sector: str) -> str:
  """Generate context-aware prompts using LangGraph state"""
  
  # Access exclusion list from state
  exclude_symbols = state.get("exclude_symbols", [])
  
  # Session awareness
  session_id = state.get("session_id")
  
  prompt = f"""Find 8-12 trending stocks in {sector} sector.
  
Session: {session_id}
Exclude: {', '.join(exclude_symbols) if exclude_symbols else 'None'}

Requirements:
- REAL publicly traded companies only
- Include varied market caps
- Focus on companies with recent catalysts
- Provide Symbol, Company Name, Founded Year, Performance
"""
  return prompt

Tool Chain Example - Market Validation Node:

def market_data_validation_agent(state: AgentState) -> AgentState:
  """Validate discoveries with real market data"""
  state["messages"].append("📊 Validating with market data")
  
  validated = []
  market_data = {}
  
  for asset in state["discovered_assets"]:
    symbol = asset.get("symbol")
    
    # Tool 1: Yahoo Finance API
    try:
      ticker = yf.Ticker(symbol)
      hist = ticker.history(period="1mo")
      
      if not hist.empty:
        # Tool 2: Price calculation
        current_price = hist['Close'].iloc[-1]
        performance = calculate_performance(hist)
        
        # Update state
        market_data[symbol] = {
          "price": current_price,
          "performance": performance
        }
        validated.append(asset)
    except Exception as e:
      state["messages"].append(f"  Failed to validate {symbol}")
  
  state["validated_assets"] = validated
  state["market_data"] = market_data
  return state

LangGraph Solution: TypedDict state with clear ownership

Problem: Different asset types require different processing paths
LangGraph Solution: Conditional edges with routing functions

  • route_emerging_type() for opportunity routing
  • should_analyze_companies() for conditional analysis

Problem: Long-running workflows failing midway, losing progress
LangGraph Solution: Built-in checkpointing with MemorySaver
  • Automatic state persistence after each node
  • Resume from last successful node on failure
  • Thread-based isolation for concurrent users
  • No manual checkpoint management needed
